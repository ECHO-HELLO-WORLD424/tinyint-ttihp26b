# Functional check at the datasheet clock (10 MHz), RTL backend.
#
#   python3 tools/chip_emulate.py repl --script tools/emulator/examples/functional_check.txt
#
# The DUT and the independent oracle must agree for every workload class and
# every delay-bank tap: zero errors at 10 MHz is the functional contract.

echo == PRBS, banks at tap 0 ==
config seg=0000 pat=prbs cansel=3 winsel=0
run 100
freeze 1
status
assert st.ops == 100
assert st.err_cnt == 0
assert st.gen_cnt > 0
assert st.mat_cnt > 0

echo == worst-case carry, longest delay banks ==
config seg=3333 pat=worst cansel=3 winsel=0
run 100
freeze 1
status
assert st.ops == 100
assert st.err_cnt == 0

echo == mixed banks, alternating operands ==
config seg=2130 pat=alt cansel=3 winsel=0
run 100
freeze 1
status
assert st.err_cnt == 0

echo == static hold (no carry activity) ==
config seg=3333 pat=hold cansel=3 winsel=0
run 100
freeze 1
status
assert st.err_cnt == 0

echo == DFT self-check: FORCE_ERR must count one error per compared op ==
config seg=0000 pat=prbs force_err=1
run 10
freeze 1
status
assert st.err_cnt == 9
assert st.err_seen
quit
